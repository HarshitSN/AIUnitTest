import pytest
import httpx
from typing import Dict
from uuid import uuid4

# Helpers
def build_pet_payload(name: str) -> Dict:
    """Create a minimal valid Pet payload for the POST /pet operation."""
    return {
        "name": name,
        "status": "available",
        "photoUrls": ["http://example.com/photo.jpg"]
    }


@pytest.mark.usefixtures("client")
def test_add_pet_happy(client, unique_pet_name, api_key_headers):
    """Happy path: add a new pet and verify basic fields, then read/update."""
    payload = build_pet_payload(unique_pet_name)

    # Create pet (requires authentication per spec)
    resp = client.post("/pet", json=payload, headers=api_key_headers)
    assert resp.status_code in (200, 201), f"Unexpected status: {resp.status_code}"
    data = resp.json()
    assert data.get("name") == unique_pet_name
    pet_id = data.get("id")
    assert isinstance(pet_id, int)

    # Read the created pet
    resp_read = client.get(f"/pet/{pet_id}", headers=api_key_headers)
    assert resp_read.status_code == 200
    assert resp_read.json().get("id") == pet_id

    # Update the pet with a new name
    updated = resp_read.json().copy()
    updated["name"] = updated.get("name", unique_pet_name) + "_updated"
    resp_update = client.put("/pet", json=updated, headers=api_key_headers)
    assert resp_update.status_code in (200, 201)

    # Read again to verify update
    resp_read_again = client.get(f"/pet/{pet_id}", headers=api_key_headers)
    assert resp_read_again.status_code == 200
    assert resp_read_again.json().get("name") == updated["name"]

import os
import pytest

try:
    import schemathesis  # type: ignore
except Exception:
    schemathesis = None  # type: ignore


SCHEMA_URL = os.environ.get("SCHEMA_URL", "https://petstore.swagger.io/v2/swagger.json")


def _schema_accessible(url: str) -> bool:
    try:
        import requests
        resp = requests.get(url, timeout=5)
        return resp.status_code == 200
    except Exception:
        return False


@pytest.mark.skipif(
    schemathesis is None or not _schema_accessible(SCHEMA_URL),
    reason="Schemathesis not installed or OpenAPI schema not accessible"
)
def test_schemathesis_property_tests():
    """Property-based tests using Schemathesis to validate endpoints against the OpenAPI schema.

    This test will generate and execute multiple cases against the Petstore API, applying an
    API key for authentication where required.
    """
    import schemathesis  # local import to satisfy optional availability
    schema = schemathesis.from_uri(SCHEMA_URL)

    @schema.parametrize()
    def test_case(case):
        """Run each generated case with authentication header and validate response schema."""
        # Attach the special API key for authentication where applicable
        case.headers = {"api_key": "special-key"}
        # Use the Petstore base URL
        case.call_and_validate(base_url="https://petstore.swagger.io/v2")

# Tests for KAN-4: User API Tests (Petstore)

This directory contains pytest-based tests for the Swagger Petstore API as per the JIRA epic KAN-4.

What is covered
- Happy path: create, read, update, and search operations on Pet endpoints
- Validation errors: invalid inputs, missing required fields, wrong data types
- Not found: 404 for missing resources
- Authentication: basic API key based auth to exercise protected endpoints
- Edge cases: status and tag filters, image upload for a pet
- Response validation: basic schema checks on responses

How to run
1) Install dependencies:
   pip install -r tests/requirements.txt

2) Run tests:
   pytest -q

3) Optional Schemathesis run:
   Set SCHEMA_URL to a public OpenAPI/Swagger URL if you want to run property tests.
   SCHEMA_URL="https://petstore.swagger.io/v2/swagger.json" pytest -q

Note
- Tests target the public Petstore API (https://petstore.swagger.io/v2).
- Some endpoints require the API key header: api_key: special-key
- The tests include a Schemathesis-based property test harness that validates endpoints against the OpenAPI schema when network access is available.
